1. Improved Str_copy Procedure
기존 Str_copy 프로시저는 복사할 문자 수를 제한하지 않는다. 복사할 최대 문자 수를 나타내는 추가 매개변수를 받는 Str_copy_n 프로시저를 작성하시오.
Str_copy_n PROC USES esi edi ecx,
    pSource:PTR BYTE,
    pTarget:PTR BYTE,
    maxChars:DWORD

    mov esi, pSource
    mov edi, pTarget
    mov ecx, maxChars

L1:
    cmp ecx, 0
    je L2

    mov al, [esi]
    cmp al, 0
    je L2

    mov [edi], al
    inc esi
    inc edi
    dec ecx
    jmp L1

L2:
    mov BYTE PTR [edi], 0
    ret

Str_copy_n ENDP

2. Str_concat Procedure
source 문자열을 target 문자열 끝에 연결하는 Str_concat 프로시저를 작성하시오.
target 문자열에는 새로운 문자를 저장할 충분한 공간이 있어야 하며, source와 target 문자열의 포인터를 전달한다.
Str_concat PROC USES esi edi,
    pTarget:PTR BYTE,
    pSource:PTR BYTE

    mov edi, pTarget

L1:
    cmp BYTE PTR [edi], 0
    je L2
    inc edi
    jmp L1

L2:
    mov esi, pSource

L3:
    mov al, [esi]
    mov [edi], al

    cmp al, 0
    je L4

    inc esi
    inc edi
    jmp L3

L4:
    ret

Str_concat ENDP
3. Str_remove Procedure
문자열에서 n개의 문자를 제거하는 Str_remove 프로시저를 작성하시오. 
제거할 위치의 포인터와 제거할 문자 수를 전달한다.
Str_remove PROC USES esi edi,
    pPosition:PTR BYTE,
    numChars:DWORD

    mov esi, pPosition
    mov edi, esi
    add esi, numChars

L1:
    mov al, [esi]
    mov [edi], al

    cmp al, 0
    je L2

    inc esi
    inc edi
    jmp L1

L2:
    ret

Str_remove ENDP
4. Str_find Procedure
source 문자열이 target 문자열 안에서 처음 일치하는 위치를 검색하는 Str_find 프로시저를 작성하시오. 
일치하면 Zero Flag를 설정하고 EAX가 target 문자열의 일치 위치를 가리키게 한다. 
일치하지 않으면 Zero Flag를 해제하고 EAX는 정의되지 않은 값이 된다.
Str_find PROC USES esi edi ebx edx,
    pSource:PTR BYTE,
    pTarget:PTR BYTE

    mov esi, pTarget
    mov edi, pSource

L1:
    cmp BYTE PTR [esi], 0
    je L5

    mov ebx, esi
    mov edx, edi

L2:
    mov al, [ebx]
    cmp al, 0
    je L3

    cmp BYTE PTR [edx], 0
    je L4

    cmp al, [edx]
    jne L4

    inc ebx
    inc edx
    jmp L2

L3:
    mov eax, esi
    cmp eax, eax
    ret

L4:
    inc esi
    cmp BYTE PTR [esi], 0
    jne L1

L5:
    mov eax, 1
    cmp eax, 0
    ret

Str_find ENDP
5. Str_nextWord Procedure
문자열에서 특정 구분 문자의 첫 번째 발생 위치를 검색하고 해당 문자를 NULL 바이트로 변경하는 Str_nextWord 프로시저를 작성하시오. 
구분 문자를 찾으면 Zero Flag를 설정하고 EAX에 구분 문자 다음 문자의 오프셋을 저장한다.
Str_nextWord PROC USES esi,
    pString:PTR BYTE,
    delimiter:BYTE

    mov esi, pString

L1:
    mov al, [esi]
    cmp al, 0
    je L2

    cmp al, delimiter
    je L3

    inc esi
    jmp L1

L2:
    mov eax, 1
    cmp eax, 0
    ret

L3:
    mov BYTE PTR [esi], 0
    inc esi
    mov eax, esi
    cmp eax, eax
    ret

Str_nextWord ENDP
6. Constructing a Frequency Table
문자열에 각 문자가 몇 번 나타나는지 계산하는 Get_frequencies 프로시저를 작성하시오. 
문자열 포인터와 0으로 초기화된 256개의 doubleword 배열 포인터를 입력받는다. 배열은 ASCII 코드로 인덱싱한다.
Get_frequencies PROC USES esi edi eax ebx,
    pString:PTR BYTE,
    pTable:PTR DWORD

    mov esi, pString
    mov edi, pTable

L1:
    mov al, [esi]
    cmp al, 0
    je L2

    movzx ebx, al
    inc DWORD PTR [edi+ebx*4]

    inc esi
    jmp L1

L2:
    ret

Get_frequencies ENDP
7. Sieve of Eratosthenes
65,000개의 byte 배열을 사용하여 2부터 65,000까지의 소수를 찾고 출력하시오. 
소수가 아닌 위치는 1로 표시하고, 배열은 STOSB를 사용하여 0으로 초기화한다.
INCLUDE Irvine32.inc

.data
primeArray BYTE 65000 DUP(?)

.code
main PROC

    mov edi, OFFSET primeArray
    mov ecx, 65000
    mov al, 0
    cld
    rep stosb

    mov ebx, 2

L1:
    mov eax, ebx
    mul ebx
    cmp eax, 65000
    jae L4

    mov esi, eax

L2:
    cmp esi, 65000
    jae L3

    mov primeArray[esi], 1
    add esi, ebx
    jmp L2

L3:
    inc ebx
    jmp L1

L4:
    mov ecx, 2

L5:
    cmp ecx, 65000
    jae L6

    cmp primeArray[ecx], 0
    jne L7

    mov eax, ecx
    call WriteDec
    call Crlf

L7:
    inc ecx
    jmp L5

L6:
    exit

main ENDP
END main
8. Bubble Sort
BubbleSort 프로시저에 교환 여부를 나타내는 변수를 추가하시오. 
내부 루프에서 값이 교환되면 변수를 1로 설정하고, 배열 전체를 한 번 검사하는 동안 교환이 없으면 정렬을 종료한다.
BubbleSort PROC USES eax ebx ecx edx esi,
    pArray:PTR DWORD,
    count:DWORD

    mov ecx, count
    dec ecx

L1:
    mov ebx, 0
    mov esi, pArray
    mov edx, 0

L2:
    cmp edx, ecx
    jae L3

    mov eax, [esi]
    cmp eax, [esi+4]
    jbe L4

    xchg eax, [esi+4]
    mov [esi], eax

    mov ebx, 1

L4:
    add esi, 4
    inc edx
    jmp L2

L3:
    cmp ebx, 0
    je L5

    dec ecx
    cmp ecx, 0
    jne L1

L5:
    ret

BubbleSort ENDP
9. Binary Search
이진 검색 프로시저를 mid, first, last를 레지스터로 사용하는 방식으로 다시 작성하고 각 레지스터의 용도를 설명하는 주석을 추가하시오.
BinarySearch PROC USES ebx ecx edx esi,
    pArray:PTR DWORD,
    count:DWORD,
    searchValue:DWORD

    mov esi, pArray

    mov ebx, 0              ; first
    mov ecx, count
    dec ecx                 ; last

L1:
    cmp ebx, ecx
    ja L4

    mov eax, ebx
    add eax, ecx
    shr eax, 1              ; mid

    mov edx, [esi+eax*4]

    cmp edx, searchValue
    je L3

    jb L2

    mov ecx, eax
    dec ecx
    jmp L1

L2:
    mov ebx, eax
    inc ebx
    jmp L1

L3:
    ret

L4:
    mov eax, -1
    ret

BinarySearch ENDP
10. Letter Matrix
모음이 선택될 확률이 50%인 무작위 대문자로 구성된 4×4 행렬을 생성하는 프로시저를 작성하시오. 
프로시저를 5번 호출하고 각 행렬을 출력하는 테스트 프로그램을 작성하시오.
INCLUDE Irvine32.inc

.data
matrix BYTE 16 DUP(?)
vowels BYTE "AEIOU"
consonants BYTE "BCDFGHJKLMNPQRSTVWXYZ"

.code
main PROC

    call Randomize

    mov ecx, 5

L1:
    push ecx

    mov esi, OFFSET matrix
    mov ecx, 16

L2:
    mov eax, 2
    call RandomRange

    cmp eax, 0
    je L3

    mov eax, 21
    call RandomRange
    mov al, consonants[eax]
    jmp L4

L3:
    mov eax, 5
    call RandomRange
    mov al, vowels[eax]

L4:
    mov [esi], al
    inc esi
    loop L2

    mov esi, OFFSET matrix
    mov ecx, 16
    mov ebx, 0

L5:
    mov al, [esi]
    call WriteChar

    inc esi
    inc ebx

    cmp ebx, 4
    jne L6

    call Crlf
    mov ebx, 0

L6:
    loop L5

    call Crlf

    pop ecx
    loop L1

    exit

main ENDP
END main
11. Letter Matrix / Sets with Vowels
4×4 문자 행렬에서 각 행, 열, 대각선을 검사하여 모음이 정확히 2개 포함된 네 글자 집합만 출력하시오.
INCLUDE Irvine32.inc

.data
matrix BYTE "POAZ"
       BYTE "AEAU"
       BYTE "GKAE"
       BYTE "IAGD"

tempSet BYTE 4 DUP(?)

.code
main PROC

    ; rows
    mov esi, OFFSET matrix
    mov ecx, 4

L1:
    mov al, [esi]
    mov tempSet[0], al
    mov al, [esi+1]
    mov tempSet[1], al
    mov al, [esi+2]
    mov tempSet[2], al
    mov al, [esi+3]
    mov tempSet[3], al

    call CheckSet

    add esi, 4
    loop L1

    ; columns
    mov ecx, 4
    mov ebx, 0

L2:
    mov al, matrix[ebx]
    mov tempSet[0], al

    mov al, matrix[ebx+4]
    mov tempSet[1], al

    mov al, matrix[ebx+8]
    mov tempSet[2], al

    mov al, matrix[ebx+12]
    mov tempSet[3], al

    call CheckSet

    inc ebx
    loop L2

    ; diagonal 1
    mov al, matrix[0]
    mov tempSet[0], al
    mov al, matrix[5]
    mov tempSet[1], al
    mov al, matrix[10]
    mov tempSet[2], al
    mov al, matrix[15]
    mov tempSet[3], al

    call CheckSet

    ; diagonal 2
    mov al, matrix[3]
    mov tempSet[0], al
    mov al, matrix[6]
    mov tempSet[1], al
    mov al, matrix[9]
    mov tempSet[2], al
    mov al, matrix[12]
    mov tempSet[3], al

    call CheckSet

    exit

main ENDP


CheckSet PROC USES esi ecx ebx eax

    mov esi, OFFSET tempSet
    mov ecx, 4
    mov ebx, 0

L1:
    mov al, [esi]

    cmp al, 'A'
    je L2
    cmp al, 'E'
    je L2
    cmp al, 'I'
    je L2
    cmp al, 'O'
    je L2
    cmp al, 'U'
    je L2

    jmp L3

L2:
    inc ebx

L3:
    inc esi
    loop L1

    cmp ebx, 2
    jne L4

    mov esi, OFFSET tempSet
    mov ecx, 4

L5:
    mov al, [esi]
    call WriteChar
    inc esi
    loop L5

    call Crlf

L4:
    ret

CheckSet ENDP

END main
12. Calculating the Sum of an Array Row
2차원 byte, word 또는 doubleword 배열의 한 행의 합을 계산하는 calc_row_sum 프로시저를 작성하시오. 배열 주소, 행 크기, 배열 타입, 행 인덱스를 스택 매개변수로 전달한다. 합은 EAX에 반환한다. 
INVOKE 또는 extended PROC를 사용하지 않고 명시적인 스택 매개변수를 사용하시오.
Calc_row_sum PROC

    push ebp
    mov ebp, esp

    push esi
    push ecx
    push ebx
    push edx

    ; [ebp+8]  = array offset
    ; [ebp+12] = row size
    ; [ebp+16] = array type (1, 2, or 4)
    ; [ebp+20] = row index

    mov esi, [ebp+8]
    mov ecx, [ebp+12]
    mov ebx, [ebp+16]
    mov edx, [ebp+20]

    mov eax, edx
    mul ecx
    mul ebx

    add esi, eax

    mov eax, 0

L1:
    cmp ecx, 0
    je L2

    cmp ebx, 1
    je L3

    cmp ebx, 2
    je L4

    add eax, DWORD PTR [esi]
    add esi, 4
    jmp L5

L3:
    movzx edx, BYTE PTR [esi]
    add eax, edx
    inc esi
    jmp L5

L4:
    movzx edx, WORD PTR [esi]
    add eax, edx
    add esi, 2

L5:
    dec ecx
    jmp L1

L2:
    pop edx
    pop ebx
    pop ecx
    pop esi

    pop ebp
    ret 16

Calc_row_sum ENDP
