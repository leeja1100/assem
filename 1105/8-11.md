8.11 Programming Exercises

1. FindLargest Procedure (최댓값 찾기)
문제: 부호 있는 DWORD 배열의 주소와 배열 길이를 매개변수로 전달받아 가장 큰 값을 EAX에 반환하는 FindLargest 프로시저를 작성하라. PROC와 PROTO를 사용하고 EAX를 제외한 변경된 레지스터를 보존하라. 음수가 포함된 서로 다른 길이의 배열 3개로 테스트하라.

답:

INCLUDE Irvine32.inc

FindLargest PROTO,
    ptrArray:PTR SDWORD,
    arraySize:DWORD

.data
arr1 SDWORD 10, -5, 30, 20
arr2 SDWORD -10, -3, -50
arr3 SDWORD 100, 200, -20, 50, 300

.code
main PROC
    INVOKE FindLargest, ADDR arr1, LENGTHOF arr1
    call WriteInt
    call Crlf

    INVOKE FindLargest, ADDR arr2, LENGTHOF arr2
    call WriteInt
    call Crlf

    INVOKE FindLargest, ADDR arr3, LENGTHOF arr3
    call WriteInt
    call Crlf

    exit
main ENDP

FindLargest PROC USES esi ecx,
    ptrArray:PTR SDWORD,
    arraySize:DWORD

    mov esi, ptrArray
    mov ecx, arraySize
    mov eax, [esi]
    add esi, 4
    dec ecx

L1:
    cmp ecx, 0
    je Done

    cmp eax, [esi]
    jge Next
    mov eax, [esi]

Next:
    add esi, 4
    dec ecx
    jmp L1

Done:
    ret
FindLargest ENDP

END main


실행 결과:
+30
-3
+300

2. Chess Board (체스판 출력)
문제: 회색과 흰색 칸이 번갈아 나타나는 8×8 체스판을 출력하라. Irvine32 라이브러리의 SetTextColor와 Gotoxy를 사용하고, 전역 변수 없이 매개변수를 사용하는 작은 프로시저로 구성하라.

답:
INCLUDE Irvine32.inc

DrawBoard PROTO
DrawSquare PROTO, row:DWORD, col:DWORD

.code
main PROC
    call Clrscr
    call DrawBoard
    exit
main ENDP

DrawBoard PROC USES esi edi
    mov esi, 0

RowLoop:
    mov edi, 0

ColLoop:
    INVOKE DrawSquare, esi, edi
    inc edi
    cmp edi, 8
    jl ColLoop

    inc esi
    cmp esi, 8
    jl RowLoop
    ret
DrawBoard ENDP

DrawSquare PROC USES eax ebx edx,
    row:DWORD,
    col:DWORD

    mov eax, row
    add eax, col
    and eax, 1

    cmp eax, 0
    je WhiteSquare

    mov eax, gray + (gray * 16)
    jmp SetColor

WhiteSquare:
    mov eax, white + (white * 16)

SetColor:
    call SetTextColor

    mov eax, col
    shl eax, 1
    mov dl, al
    mov eax, row
    mov dh, al
    call Gotoxy

    mov al, ' '
    call WriteChar
    call WriteChar
    ret
DrawSquare ENDP

END main


실행 결과: 회색과 흰색 칸이 번갈아 배치된 8×8 체스판이 출력된다.

3. Chess Board with Alternating Colors (색상이 바뀌는 체스판)
문제: 2번 문제를 확장하여 500밀리초마다 색상 칸의 배경색을 변경하고 체스판을 다시 출력하라. 16가지 배경색을 모두 표시하며 흰색 칸은 그대로 유지하라.

답:
INCLUDE Irvine32.inc

DrawBoard PROTO, bgColor:DWORD
DrawSquare PROTO,
    row:DWORD,
    col:DWORD,
    bgColor:DWORD

.code
main PROC
    mov ebx, 0

L1:
    call Clrscr
    INVOKE DrawBoard, ebx

    mov eax, 500
    call Delay

    inc ebx
    cmp ebx, 16
    jl L1

    mov eax, lightGray + (black * 16)
    call SetTextColor
    exit
main ENDP

DrawBoard PROC USES esi edi,
    bgColor:DWORD

    mov esi, 0

RowLoop:
    mov edi, 0

ColLoop:
    INVOKE DrawSquare, esi, edi, bgColor
    inc edi
    cmp edi, 8
    jl ColLoop

    inc esi
    cmp esi, 8
    jl RowLoop
    ret
DrawBoard ENDP

DrawSquare PROC USES eax edx,
    row:DWORD,
    col:DWORD,
    bgColor:DWORD

    mov eax, row
    add eax, col
    and eax, 1

    cmp eax, 0
    je WhiteSquare

    mov eax, bgColor
    jmp SetColor

WhiteSquare:
    mov eax, white

SetColor:
    shl eax, 4
    call SetTextColor

    mov eax, col
    shl eax, 1
    mov dl, al
    mov eax, row
    mov dh, al
    call Gotoxy

    mov al, ' '
    call WriteChar
    call WriteChar
    ret
DrawSquare ENDP

END main


실행 결과: 체스판의 색상이 0.5초마다 변경되며 총 16가지 배경색이 표시된다.
4. FindThrees Procedure (연속된 3 찾기)
문제: 배열에 숫자 3이 연속으로 세 번 나타나면 EAX에 1을 반환하고, 그렇지 않으면 0을 반환하는 FindThrees 프로시저를 작성하라. 배열 주소와 크기를 매개변수로 전달하고 여러 배열로 테스트하라.

답:
INCLUDE Irvine32.inc

FindThrees PROTO,
    ptrArray:PTR DWORD,
    arraySize:DWORD

.data
arr1 DWORD 1, 3, 3, 3, 5
arr2 DWORD 3, 2, 3, 3, 1
arr3 DWORD 3, 3, 3, 3

.code
main PROC
    INVOKE FindThrees, ADDR arr1, LENGTHOF arr1
    call WriteDec
    call Crlf

    INVOKE FindThrees, ADDR arr2, LENGTHOF arr2
    call WriteDec
    call Crlf

    INVOKE FindThrees, ADDR arr3, LENGTHOF arr3
    call WriteDec
    call Crlf

    exit
main ENDP

FindThrees PROC USES esi ecx,
    ptrArray:PTR DWORD,
    arraySize:DWORD

    mov esi, ptrArray
    mov ecx, arraySize
    xor eax, eax

    cmp ecx, 3
    jb Done

    sub ecx, 2

L1:
    cmp DWORD PTR [esi], 3
    jne Next
    cmp DWORD PTR [esi+4], 3
    jne Next
    cmp DWORD PTR [esi+8], 3
    jne Next

    mov eax, 1
    jmp Done

Next:
    add esi, 4
    loop L1

Done:
    ret
FindThrees ENDP

END main


실행 결과:
1
0
1


5. DifferentInputs Procedure (서로 다른 입력값 판별)
문제: 세 개의 매개변수 값이 모두 다르면 EAX에 1을 반환하고, 하나라도 같으면 0을 반환하는 DifferentInputs 프로시저를 작성하라. PROC와 PROTO를 사용하고 서로 다른 입력으로 5번 테스트하라.

답:
INCLUDE Irvine32.inc

DifferentInputs PROTO,
    a:DWORD,
    b:DWORD,
    c:DWORD

.code
main PROC
    INVOKE DifferentInputs, 1, 2, 3
    call WriteDec
    call Crlf

    INVOKE DifferentInputs, 1, 1, 3
    call WriteDec
    call Crlf

    INVOKE DifferentInputs, 2, 3, 2
    call WriteDec
    call Crlf

    INVOKE DifferentInputs, 5, 5, 5
    call WriteDec
    call Crlf

    INVOKE DifferentInputs, 10, 20, 30
    call WriteDec
    call Crlf

    exit
main ENDP

DifferentInputs PROC USES ebx,
    a:DWORD,
    b:DWORD,
    c:DWORD

    mov eax, a
    mov ebx, b

    cmp eax, ebx
    je Same

    cmp eax, c
    je Same

    cmp ebx, c
    je Same

    mov eax, 1
    jmp Done

Same:
    mov eax, 0

Done:
    ret
DifferentInputs ENDP

END main


실행 결과:
1
0
0
0
1



6. Exchanging Integers (정수 교환)
문제: 무작위 순서로 배열된 정수 배열을 생성하라. 8.4.6절의 Swap 프로시저를 사용하여 반복문으로 배열의 인접한 두 정수를 차례로 교환하라.

답:
INCLUDE Irvine32.inc

Swap PROTO,
    p1:PTR DWORD,
    p2:PTR DWORD

.data
arr DWORD 10, 20, 30, 40, 50

.code
main PROC
    mov esi, OFFSET arr
    mov ecx, LENGTHOF arr - 1

L1:
    push ecx
    mov ebx, esi
    add ebx, 4

    INVOKE Swap, esi, ebx

    add esi, 4
    pop ecx
    loop L1

    mov esi, OFFSET arr
    mov ecx, LENGTHOF arr

L2:
    mov eax, [esi]
    call WriteDec
    call Crlf
    add esi, 4
    loop L2

    exit
main ENDP

Swap PROC USES eax ebx,
    p1:PTR DWORD,
    p2:PTR DWORD

    mov esi, p1
    mov edi, p2

    mov eax, [esi]
    mov ebx, [edi]

    mov [esi], ebx
    mov [edi], eax
    ret
Swap ENDP

END main


실행 결과:
20
30
40
50
10


7. Greatest Common Divisor (최대공약수)
문제: 유클리드 호제법을 재귀적으로 구현하여 두 정수의 최대공약수(GCD)를 구하는 프로시저를 작성하라. 다음 다섯 쌍을 테스트하고 각 결과를 출력하라.

(5, 20), (24, 18), (11, 7), (432, 226), (26, 13)

답:
INCLUDE Irvine32.inc

GCD PROTO,
    a:DWORD,
    b:DWORD

.code
main PROC
    INVOKE GCD, 5, 20
    call WriteDec
    call Crlf

    INVOKE GCD, 24, 18
    call WriteDec
    call Crlf

    INVOKE GCD, 11, 7
    call WriteDec
    call Crlf

    INVOKE GCD, 432, 226
    call WriteDec
    call Crlf

    INVOKE GCD, 26, 13
    call WriteDec
    call Crlf

    exit
main ENDP

GCD PROC USES ebx edx,
    a:DWORD,
    b:DWORD

    mov eax, a
    mov ebx, b

    cmp ebx, 0
    je Done

    xor edx, edx
    div ebx

    INVOKE GCD, ebx, edx

Done:
    ret
GCD ENDP

END main

8. Counting Matching Elements (일치하는 원소 개수 세기)
문제: 두 개의 부호 있는 DWORD 배열 주소와 배열 길이를 전달받는 CountMatches 프로시저를 작성하라. 두 배열의 같은 위치에 있는 원소가 같으면 개수를 1 증가시키고 최종 개수를 EAX에 반환하라. INVOKE, PROTO를 사용하고 두 쌍의 배열로 테스트하라. EAX를 제외한 변경된 레지스터는 보존하라.

답:
INCLUDE Irvine32.inc

CountMatches PROTO,
    p1:PTR SDWORD,
    p2:PTR SDWORD,
    count:DWORD

.data
arr1 SDWORD 1, 2, 3, 4, 5
arr2 SDWORD 1, 0, 3, 0, 5

arr3 SDWORD 10, 20, 30, 40
arr4 SDWORD 10, 20, 0, 0

.code
main PROC
    INVOKE CountMatches, ADDR arr1, ADDR arr2, LENGTHOF arr1
    call WriteDec
    call Crlf

    INVOKE CountMatches, ADDR arr3, ADDR arr4, LENGTHOF arr3
    call WriteDec
    call Crlf

    exit
main ENDP

CountMatches PROC USES esi edi ebx ecx,
    p1:PTR SDWORD,
    p2:PTR SDWORD,
    count:DWORD

    mov esi, p1
    mov edi, p2
    mov ecx, count
    xor eax, eax

L1:
    cmp ecx, 0
    je Done

    mov ebx, [esi]
    cmp ebx, [edi]
    jne Next

    inc eax

Next:
    add esi, 4
    add edi, 4
    dec ecx
    jmp L1

Done:
    ret
CountMatches ENDP

END main


실행 결과:
3
2


9. Counting Nearly Matching Elements (거의 일치하는 원소 개수 세기)
문제: 두 개의 부호 있는 DWORD 배열 주소, 배열 길이, 허용 가능한 최대 차이(diff)를 매개변수로 받는 CountNearMatches 프로시저를 작성하라. 같은 위치의 두 원소 차이의 절댓값이 diff 이하이면 개수를 증가시키고 EAX에 반환하라. 두 쌍의 배열로 테스트하고 PROTO, INVOKE를 사용하라. EAX를 제외한 변경된 레지스터는 보존하라.

답:
INCLUDE Irvine32.inc

CountNearMatches PROTO,
    p1:PTR SDWORD,
    p2:PTR SDWORD,
    count:DWORD,
    diff:DWORD

.data
arr1 SDWORD 10, 20, 30, 40
arr2 SDWORD 12, 25, 29, 50

arr3 SDWORD -5, 10, 20
arr4 SDWORD -3, 15, 18

.code
main PROC
    INVOKE CountNearMatches, ADDR arr1, ADDR arr2, 4, 3
    call WriteDec
    call Crlf

    INVOKE CountNearMatches, ADDR arr3, ADDR arr4, 3, 2
    call WriteDec
    call Crlf

    exit
main ENDP

CountNearMatches PROC USES esi edi ebx ecx edx,
    p1:PTR SDWORD,
    p2:PTR SDWORD,
    count:DWORD,
    diff:DWORD

    mov esi, p1
    mov edi, p2
    mov ecx, count
    xor eax, eax

L1:
    cmp ecx, 0
    je Done

    mov ebx, [esi]
    sub ebx, [edi]

    jns Positive
    neg ebx

Positive:
    cmp ebx, diff
    ja Next

    inc eax

Next:
    add esi, 4
    add edi, 4
    dec ecx
    jmp L1

Done:
    ret
CountNearMatches ENDP

END main


실행 결과:
2
2


10. Show Procedure Parameters (프로시저 매개변수 출력)
문제: 호출한 프로시저의 런타임 스택에 저장된 32비트 매개변수의 주소와 16진수 값을 출력하는 ShowParams 프로시저를 작성하라. 출력할 매개변수 개수를 하나의 정수로 전달받는다. 매개변수는 낮은 주소부터 높은 주소 순서로 출력한다.

예를 들어 다음과 같이 호출한다.
INVOKE MySample, 1234h, 5000h, 6543h


MySample 내부에서 ShowParams를 호출하여 세 매개변수의 주소와 값을 출력하라.

답:
INCLUDE Irvine32.inc

MySample PROTO,
    first:DWORD,
    second:DWORD,
    third:DWORD

ShowParams PROTO,
    paramCount:DWORD

.data
msg1 BYTE "Stack parameters:", 0
msg2 BYTE "Address ", 0
msg3 BYTE " = ", 0

.code
main PROC
    INVOKE MySample, 1234h, 5000h, 6543h
    exit
main ENDP

MySample PROC,
    first:DWORD,
    second:DWORD,
    third:DWORD

    INVOKE ShowParams, 3
    ret
MySample ENDP

ShowParams PROC USES esi ebx ecx,
    paramCount:DWORD

    mov edx, OFFSET msg1
    call WriteString
    call Crlf

    mov esi, ebp
    mov esi, [esi]
    add esi, 8

    mov ecx, paramCount

L1:
    cmp ecx, 0
    je Done

    mov edx, OFFSET msg2
    call WriteString

    mov eax, esi
    call WriteHex

    mov edx, OFFSET msg3
    call WriteString

    mov eax, [esi]
    call WriteHex
    call Crlf

    add esi, 4
    dec ecx
    jmp L1

Done:
    ret
ShowParams ENDP

END main


실행 결과 예:
Stack parameters:
Address 0012FF80 = 00001234
Address 0012FF84 = 00005000
Address 0012FF88 = 00006543
