INCLUDE Irvine32.inc

.data
    msg1 BYTE "Enter first number: ", 0
    msg2 BYTE "Enter second number: ", 0

    msgAdd BYTE "Addition = ", 0
    msgSub BYTE "Subtraction = ", 0
    msgMul BYTE "Multiplication = ", 0
    msgDiv BYTE "Division = ", 0

    msgAX BYTE "AX = ", 0
    msgBX BYTE "BX = ", 0
    msgCX BYTE "CX = ", 0
    msgDX BYTE "DX = ", 0

    num1 DWORD ?
    num2 DWORD ?

.code
main PROC

    ; 첫 번째 숫자 입력
    mov edx, OFFSET msg1
    call WriteString
    call ReadInt
    mov num1, eax

    ; 두 번째 숫자 입력
    mov edx, OFFSET msg2
    call WriteString
    call ReadInt
    mov num2, eax

    ; 덧셈
    mov eax, num1
    add eax, num2
    mov edx, OFFSET msgAdd
    call WriteString
    call WriteInt
    call Crlf

    ; 뺄셈
    mov eax, num1
    sub eax, num2
    mov edx, OFFSET msgSub
    call WriteString
    call WriteInt
    call Crlf

    ; 곱셈
    mov eax, num1
    mov ebx, num2
    mul ebx
    mov edx, OFFSET msgMul
    call WriteString
    call WriteInt
    call Crlf

    ; 나눗셈
    mov eax, num1
    mov ebx, num2
    cdq
    idiv ebx
    mov edx, OFFSET msgDiv
    call WriteString
    call WriteInt
    call Crlf

    ; AX 출력
    mov edx, OFFSET msgAX
    call WriteString
    movzx eax, ax
    call WriteInt
    call Crlf

    ; BX 출력
    mov edx, OFFSET msgBX
    call WriteString
    movzx eax, bx
    call WriteInt
    call Crlf

    ; CX 출력
    mov edx, OFFSET msgCX
    call WriteString
    movzx eax, cx
    call WriteInt
    call Crlf

    ; DX 출력
    mov edx, OFFSET msgDX
    call WriteString
    movzx eax, dx
    call WriteInt
    call Crlf

    exit
main ENDP
END main
