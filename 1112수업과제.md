INCLUDE Irvine32.inc

.data
    msg1 BYTE "Enter 5 integers:", 0
    msg2 BYTE "Average = ", 0

    sum DWORD 0
    avg DWORD 0

.code
main PROC

    mov edx, OFFSET msg1
    call WriteString
    call Crlf

    mov ecx, 5

L1:
    call ReadInt
    add sum, eax
    loop L1

    mov eax, sum
    mov ebx, 5
    cdq
    idiv ebx

    mov avg, eax

    mov edx, OFFSET msg2
    call WriteString

    mov eax, avg
    call WriteInt
    call Crlf

    exit

main ENDP
END main
